\partslide{Model Theory}

\begin{frame}{What is Model Theory?}
  \begin{itemize}
    \item Model theory is a branch of mathematical logic where we study mathematical structures by considering the first-order sentences true in those structures and the sets definable by first-order formulas.
    \item Given concrete mathematical structures, we can obtain new information about the structure using model-theoretic techniques.
  \end{itemize}
\end{frame}

\begin{frame}{Language}
  \begin{itemize}
    \item In mathematical logic, we use first-order languages to describe mathematical structures.
    \item A $\mathit{language}$ $\mathcal{L}$ consists of:
      \begin{itemize}
        \item a set of function symbols $\mathcal{F}$
        \item a set of relation symbols $\mathcal{R}$
        \item a set of constant symbols $\mathcal{C}$
      \end{itemize}
    \item Examples:
      \begin{itemize}
        \item The language of rings $\mathcal{L}_r = \{+, -, \cdot, 0, 1\}$
        \item The language of orders $\mathcal{L}_o = \{<\}$
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Structure}
  \begin{itemize}
    \item A structure interprets the symbols of a language on a domain.
    \item Formally, an \text{$\mathcal{L}$-structure} $\mathcal{M}$ consists of:
      \begin{itemize}
        \item a non-empty set \(M\) (the domain/universe),
        \item for each function symbol \(f \in \mathcal{F}\), a function \(f^{\mathcal{M}} : \mathcal{M}^n \to \mathcal{M}\),
        \item for each relation symbol \(R \in \mathcal{R}\), a relation \(R^{\mathcal{M}} \subseteq \mathcal{M}^n\),
        \item for each constant symbol \(c \in \mathcal{C}\), an element \(c^{\mathcal{M}} \in \mathcal{M}\).
      \end{itemize}
    \item Examples: \((\mathbb{N}, 0, 1, +, \cdot)\), {\((\mathbb{Q}, \leq)\)}.
  \end{itemize}
\end{frame}

\begin{frame}{Terms}
  \begin{itemize}
    \item An \textit{$\mathcal{L}$-term} is:
    \begin{itemize}
      \item a constant symbol \(c\),
      \item a variable \(v_i\)
      \item or a function symbol \(f\) applied to terms \(t_1, \ldots, t_n\).
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Formula and Sentence}
  \begin{itemize}
    \item An \textit{atomic $\mathcal{L}$-formula} is a relation symbol applied to terms, or an equality between terms.
    \item \textit{$\mathcal{L}$-formulas} are built from atomic formulas using logical connectives and quantifiers.
    \item An \textit{$\mathcal{L}$-sentence} is a formula with no free variables.
    \item Example: \(\exists z\,(x < z \wedge z < y)\) is formula but not a sentence, as it has free variables \(x\) and \(y\).
  \end{itemize}
\end{frame}

\begin{frame}{Satisfaction}
  \begin{itemize}
    \item We say that an $\mathcal{M}$ \textit{satisfies} a formula $\varphi$, written $\mathcal{M} \models \varphi$, if $\varphi$ holds in $\mathcal{M}$.
    \item Formally, the satisfaction relation is defined inductively on the structure of formulas.
    \item Example: \((\mathbb{Q}, <) \models \forall x \forall y\, (x<y \implies \exists z\, x < z \wedge z < y)\).
  \end{itemize}
\end{frame}

\begin{frame}{Theory and Model}
  \begin{itemize}
    \item An \textit{$\mathcal{L}$-theory} is a set of $\mathcal{L}$-sentences.
    \item We say that a structure $\mathcal{M}$ is a \textit{model} of a theory $T$, written $\mathcal{M} \models T$, if $\mathcal{M}$ satisfies every sentence in $T$.
    \item A theory is \textit{satisfiable} if it has a model.
    \item Moreover, we say that a theory $T$ satisfies a sentence $\varphi$, written $T \models \varphi$, if every model of $T$ satisfies $\varphi$.
  \end{itemize}
\end{frame}

\begin{frame}{Quantifier Elimination}
  \begin{itemize}
    \item A theory $T$ has \textit{quantifier elimination} if for every formula $\varphi$, there is a quantifier-free formula $\psi$ such that

    \[
    T \models \varphi \leftrightarrow \psi
    \]
  \end{itemize}
\end{frame}

\begin{frame}{Dense Linear Orders without Endpoints}
  \small
  \begin{itemize}
    \item Let \(\mathcal{L}_o = \{<\}\) be the language of orders.
    \item Let DLO be the $\mathcal{L}_o$-theory consisting of the following sentences:
      \[
      \begin{array}{@{}l@{\qquad}l@{}}
        \forall x\, \neg (x < x) &
          \text{(Irreflexivity)} \\
        \forall x\,\forall y\,\forall z\, (x < y \wedge y < z \to x < z) &
          \text{(Transitivity)} \\
        \forall x\,\forall y\, (x < y \vee x = y \vee y < x) &
          \text{(Linearity)} \\
        \forall x\,\forall y\, (x < y \to \exists z\, (x < z \wedge z < y)) &
          \text{(Density)} \\
        \forall x\,\exists y\, \exists z (y < x \wedge x < z) &
          \text{(No endpoints)}
      \end{array}
      \]
    \item Examples of models of DLO: \((\mathbb{Q}, <)\), \((\mathbb{R}, <)\), \(((0, 1), <)\) with the usual order.
  \end{itemize}
\end{frame}

\begin{frame}{Does DLO have Quantifier Elimination?}
  \small
  \begin{itemize}
    \item DLO has quantifier elimination.
    \item Every formula in the language \(\{<\}\) is equivalent over DLO to a quantifier-free formula.
    \item Examples:
        \[
      \begin{array}{@{}rcl@{}}
        \exists z\, (x < z \wedge z < y)
          &\Longleftrightarrow& x < y \\[0.6em]
        \exists z\, (x < z \wedge w < z \wedge z < y)
          &\Longleftrightarrow& x < y \wedge w < y \\[0.6em]
        \exists z\, \bigl(x < z \wedge z < y \wedge \neg(z < w)\bigr)
          &\Longleftrightarrow& x < y \wedge (w < y \vee w = y)
      \end{array}
      \]
  \end{itemize}
\end{frame}


\partslide{Type Theory}

\begin{frame}{Judgements}
  \begin{itemize}
    \item \(a : A\) is a judgement: \(a\) has type \(A\).
    \item Every expression has a type.
    \item Definitional equality is written \(a \equiv b\).
  \end{itemize}
\end{frame}

\begin{frame}{Propositions as Types}
  \begin{itemize}
    \item A proposition is viewed as a type of evidence.
    \item A proof of \(P\) is a term \(p : P\).
    \item To prove a theorem is to construct an object of the corresponding type.
  \end{itemize}
\end{frame}

\begin{frame}{Constructive Reasoning}
  \begin{itemize}
    \item A proof of \(\exists x, P(x)\) gives a witness and evidence.
    \item A proof of \(P \vee Q\) chooses one side and proves it.
    \item The law of excluded middle, \(P \vee \neg P\), is not built in constructively.
    \item Lean allows classical reasoning when explicitly invoked.
  \end{itemize}
\end{frame}

\begin{frame}{Curry--Howard Isomorphism}
  \[
  \begin{aligned}
  \interp{P \Rightarrow Q} &\equiv \interp{P} \to \interp{Q} \\
  \interp{P \wedge Q} &\equiv \interp{P} \times \interp{Q} \\
  \interp{\mathrm{True}} &\equiv 1 \\
  \interp{P \vee Q} &\equiv \interp{P} + \interp{Q} \\
  \interp{\mathrm{False}} &\equiv 0 \\
  \interp{\forall x : A.\,P} &\equiv \Pi x : A.\,\interp{P} \\
  \interp{\exists x : A.\,P} &\equiv \Sigma x : A.\,\interp{P}
  \end{aligned}
  \]
\end{frame}

\partslide{Formalization in Lean}

\begin{frame}[fragile]
\begin{Verbatim}[fontsize=\scriptsize,commandchars=\\\|\@]
\lcom|/-- A formula is equivalent to a quantifier-free formula over a theory. -/@
\lkw|def@ \ldecl|IsQFEquivalent@ (T : L.Theory) {α : \ltype|Type*@} (φ : L.Formula α) : \ltype|Prop@ :=
  \lop|∃@ ψ : L.Formula α, ψ.IsQF \lop|∧@ φ \lop|⇔[T]@ ψ

\lcom|/-- A theory has quantifier elimination if every formula is equivalent,@
\lcom|over the theory, to a quantifier-free formula. -/@
\lkw|def@ \ldecl|HasQuantifierElimination@ (T : L.Theory) : \ltype|Prop@ :=
  \lop|∀@ {α : \ltype|Type@} (φ : L.Formula α), T.IsQFEquivalent φ
\end{Verbatim}
\end{frame}
